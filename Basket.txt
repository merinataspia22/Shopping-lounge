<script>
    const products = [
        { id: 1, name: 'Product 1', image: 'product1.jpg' },
        { id: 2, name: 'Product 2', image: 'product2.jpg' },
        { id: 3, name: 'Product 3', image: 'product3.jpg' }
    ];
    const basket = [];

    function renderProducts() {
        const productShowcase = document.getElementById('productShowcase');
        productShowcase.innerHTML = '';
        products.forEach(product => {
            const productItem = document.createElement('div');
            productItem.classList.add('product-item');
            productItem.innerHTML = `
                <img src="${product.image}" alt="${product.name}">
                <div>
                    <h3>${product.name}</h3>
                    <button onclick="addToBasket(${product.id})">Add to Basket</button>
                </div>
            `;
            productShowcase.appendChild(productItem);
        });
    }

    function addToBasket(productId) {
        const product = products.find(p => p.id === productId);
        basket.push(product);
        renderBasket();
    }

    function renderBasket() {
        const basketContainer = document.getElementById('basket');
        basketContainer.innerHTML = '';
        basket.forEach(product => {
            const basketItem = document.createElement('div');
            basketItem.classList.add('basket-item');
            basketItem.innerHTML = `
                <img src="${product.image}" alt="${product.name}">
                <p>${product.name}</p>
            `;
            basketContainer.appendChild(basketItem);
        });
    }

    // Initial rendering
    renderProducts();
    renderBasket();
</script>
