<script>
    let cart = [];
    
    function addToCart() {
        // Add product to cart (dummy implementation)
        cart.push({
            name: "Product 1",
            price: 10
        });
        updateOrderSummary();
    }

    function updateOrderSummary() {
        let orderList = document.getElementById('orderList');
        let totalPrice = document.getElementById('totalPrice');
        let totalItems = document.getElementById('totalItems');
        
        // Clear existing order list
        orderList.innerHTML = '';
        
        // Populate order list with cart items
        cart.forEach(item => {
            let listItem = document.createElement('li');
            listItem.textContent = item.name + ' - $' + item.price.toFixed(2);
            orderList.appendChild(listItem);
        });
        
        // Update total price and total items
        let price = cart.reduce((acc, item) => acc + item.price, 0);
        totalPrice.textContent = price.toFixed(2);
        totalItems.textContent = cart.length;
    }

    function checkout() {
        // Dummy checkout process (clears cart and provides feedback)
        cart = [];
        updateOrderSummary();
        document.getElementById('checkoutFeedback').textContent = 'Checkout successful! Thank you for your order.';
    }
</script>
